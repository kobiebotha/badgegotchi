"""Dots & Boxes: two humans, one Tufty 2350."""

import badgeware  # Installs the badge, screen, font, and input builtins.
from time import ticks_ms, ticks_diff

from controls import Controller
from view import View


badge.mode(HIRES | VSYNC)
controller = Controller()
view = View(screen, color, font)
last_tick = ticks_ms()
buttons = (
    (BUTTON_A, "left"),
    (BUTTON_C, "right"),
    (BUTTON_UP, "up"),
    (BUTTON_DOWN, "down"),
    (BUTTON_B, "b"),
)


def update():
    global last_tick
    now = ticks_ms()
    elapsed = ticks_diff(now, last_tick)
    last_tick = now
    held = [name for button, name in buttons if badge.held(button)]
    controller.step(held, elapsed)
    view.draw(controller)


# Badgeware imports apps. Its loop also handles HOME and display updates.
run(update)
